cmake_minimum_required(VERSION 3.20)
project(KeyLayoutSwitch VERSION 1.4.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(KeyLayoutSwitch WIN32 src/App.cpp app.rc)
target_compile_definitions(KeyLayoutSwitch PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(KeyLayoutSwitch PRIVATE comctl32 imm32 shell32 ole32 oleaut32 uuid taskschd advapi32 gdi32 user32 gdiplus dwmapi)
if(MSVC)
  target_compile_options(KeyLayoutSwitch PRIVATE /utf-8 /W4)
  set_property(TARGET KeyLayoutSwitch PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
  target_link_options(KeyLayoutSwitch PRIVATE -municode -static)
endif()
enable_testing()
add_executable(CoreTests tests/CoreTests.cpp)
target_compile_definitions(CoreTests PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(CoreTests PRIVATE imm32 user32)
if(MSVC)
  target_compile_options(CoreTests PRIVATE /utf-8 /W4)
elseif(MINGW)
  target_link_options(CoreTests PRIVATE -static)
endif()
add_test(NAME KeyboardCore COMMAND CoreTests)
add_executable(EngineTests tests/EngineTests.cpp)
target_compile_definitions(EngineTests PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(EngineTests PRIVATE comctl32 imm32 shell32 ole32 oleaut32 uuid taskschd advapi32 gdi32 user32 gdiplus dwmapi)
if(MSVC)
  target_compile_options(EngineTests PRIVATE /utf-8 /W4)
elseif(MINGW)
  target_link_options(EngineTests PRIVATE -static)
endif()
add_test(NAME KeyboardWorker COMMAND EngineTests)
add_executable(StartupTests tests/StartupTests.cpp)
target_compile_definitions(StartupTests PRIVATE UNICODE _UNICODE)
target_link_libraries(StartupTests PRIVATE ole32 oleaut32 uuid taskschd advapi32)
if(MSVC)
  target_compile_options(StartupTests PRIVATE /utf-8 /W4)
elseif(MINGW)
  target_link_options(StartupTests PRIVATE -static)
endif()
add_test(NAME ElevatedStartupDefinition COMMAND StartupTests)
